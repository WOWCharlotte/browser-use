# 需求思维导图

```mermaid
tree
root[Agent 控制功能前端实现]
  backend[后端 API]
    pause[/agent/pause POST]
    resume[/agent/resume POST]
    stop[/agent/stop POST]
    status[/agent/status/{session_id} GET]
    session[Session ID 关联]
  frontend[前端实现]
    api[API 客户端]
      pauseAgent[pauseAgent(sessionId)]
      resumeAgent[resumeAgent(sessionId)]
      stopAgent[stopAgent(sessionId)]
      getAgentStatus[getAgentStatus(sessionId)]
    component[AgentControlBar 组件]
      status[状态显示]
        running[Running - 绿色]
        paused[Paused - 黄色]
        stopped[Stopped - 红色]
      controls[控制按钮]
        pause[暂停按钮]
        resume[恢复按钮]
        stop[停止按钮]
      polling[状态轮询]
        interval[1秒轮询]
    integration[集成到页面]
      location[位于 ChatWindow 上方]
      state[传递 sessionId]
  user[用户交互]
    view[查看状态]
    pause[点击暂停]
    resume[点击恢复]
    stop[点击停止]
  events[SSE 事件流]
    done[done 事件]
    paused[paused 事件]
    error[error 事件]
```
